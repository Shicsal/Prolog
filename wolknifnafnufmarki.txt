person(volk).
person(nif).
person(naf).
person(nuf).

topic(flora).
topic(sport).
topic(cosmos).
topic(fauna).

left_of(P1, P2) :-
    ((P2 =:= 1, P1 =:= 4);
     (P2 > 1, P1 =:= P2 - 1)).

right_of(P1, P2) :-
    left_of(P2, P1).

opposite(P1, P2) :-
    (P1 =:= 1, P2 =:= 3);
    (P1 =:= 2, P2 =:= 4);
    (P1 =:= 3, P2 =:= 1);
    (P1 =:= 4, P2 =:= 2).

solve(Assignments) :-
    permutation([1,2,3,4], [VolkPos, NifPos, NafPos, NufPos]),
    
    Topics = [VolkTopic, NifTopic, NafTopic, NufTopic],
    permutation([flora, sport, cosmos, fauna], Topics),
    
    VolkTopic = fauna,

    left_of(VolkPos, NafPos),
    
    nth0(Idx, Topics, cosmos),
    nth0(Idx, [VolkPos, NifPos, NafPos, NufPos], CosmosPos),
    right_of(NifPos, CosmosPos),
    
    opposite(NufPos, NafPos),
    
    NufTopic \= sport,
    
    Assignments = [
        volk-VolkPos-VolkTopic,
        nif-NifPos-NifTopic,
        naf-NafPos-NafTopic,
        nuf-NufPos-NufTopic
    ],
    !.

print_solution :-
    (solve(Assignments) ->
        print_assignments(Assignments)
    ;
        format('Решение не найдено~n', [])
    ).

print_assignments([]).
print_assignments([Person-Pos-Topic|Rest]) :-
    format('  ~w: позиция ~w, марки ~w~n', [Person, Pos, Topic]),
    print_assignments(Rest).

:- initialization(print_solution).
